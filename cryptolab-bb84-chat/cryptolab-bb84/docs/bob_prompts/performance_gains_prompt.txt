```python
## performance_gains/benchmark.py
"""Benchmark simulation methods across qubit counts."""
import time
import pandas as pd
from tqdm import tqdm
import streamlit as st
from fast.bb84_vectorised import run_bb84_vectorised
from qkd.bb84_density import run_bb84_density
from qkd.bb84_core import run_bb84

def run_benchmark(qubit_range, methods, repeats=3, use_caching=False):
    """
    Run benchmarks for given methods and qubit counts.
    Returns DataFrame with columns: qubits, method, latency_ms, repetition.
    """
    results = []
    total = len(qubit_range) * len(methods) * repeats
    progress = tqdm(total=total, desc="Benchmarking")
    
    for qubits in qubit_range:
        for method in methods:
            for rep in range(repeats):
                if method == "vectorised":
                    func = run_bb84_vectorised
                elif method == "density":
                    func = run_bb84_density
                else:
                    func = run_bb84
                start = time.perf_counter()
                func(qubits, 0.0, "none")
                end = time.perf_counter()
                latency_ms = (end - start) * 1000
                results.append({
                    "qubits": qubits,
                    "method": method,
                    "repetition": rep,
                    "latency_ms": latency_ms
                })
                progress.update(1)
    progress.close()
    df = pd.DataFrame(results)
    # Aggregate by median
    summary = df.groupby(["qubits", "method"])["latency_ms"].agg(["median", "std"]).reset_index()
    summary.columns = ["qubits", "method", "median_latency_ms", "std_latency_ms"]
    return df, summary

# Add a cached version to avoid repeated benchmarks
_benchmark_cache = {}
def get_benchmark(qubit_range, methods, repeats=3, force_refresh=False):
    key = (tuple(qubit_range), tuple(methods), repeats)
    if force_refresh or key not in _benchmark_cache:
        _benchmark_cache[key] = run_benchmark(qubit_range, methods, repeats)
    return _benchmark_cache[key]
```
